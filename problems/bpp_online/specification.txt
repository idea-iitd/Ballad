def get_valid_bin_indices(item: float, bins: np.ndarray) -> np.ndarray:
    """Returns indices of bins in which the item can fit."""
    return np.nonzero((bins - item) >= 0)[0]

def is_valid_packing(packing: list[list[float]], instance: dict) -> bool:
    """
    Checks whether `packing` is a valid packing for the given instance.

    A packing is valid if:
    - Every item in instance["items"] appears exactly once across all bins.
    - No bin exceeds instance["capacity"].

    Args:
        packing : list[list[float]]
            List of bins, where each bin is a list of item sizes.
        instance : dict
            Dictionary with at least 'items' and 'capacity' keys.

    Returns:
        bool : True if packing is valid, False otherwise.
    """
    items = instance["items"]
    capacity = instance["capacity"]

    # Flatten all packed items
    packed_items = [x for bin_items in packing for x in bin_items]

    # Check all items are packed exactly once
    if sorted(packed_items) != sorted(items):
        return False

    # Check that no bin exceeds capacity
    for bin_items in packing:
        if sum(bin_items) > capacity + 1e-8:  # tolerance for floating-point precision
            return False

    return True

@funsearch.evolve
def priority(item: float, bins_remain_cap: np.ndarray) -> np.ndarray:
  """
    Computes a priority score for each bin in an Online Bin Packing problem.

    In the Online Bin Packing problem, items arrive sequentially and must be placed
    into bins with limited capacity. This `priority` function assigns a heuristic
    score to each bin, indicating how suitable it is for accommodating the current item.
    The bin with the highest priority score is selected for placement.

    Args:
        item : float
            The size of the current item to be packed.
        bins_remain_cap : np.ndarray
            A 1D array where each element represents the remaining capacity
            of a corresponding bin.

    Returns:
        np.ndarray
            A 1D array of priority scores for all bins. Each score reflects the
            relative suitability of placing the current item in that bin.
            The bin with the highest score is considered the best choice.
  """
  max_bin_cap = max(bins_remain_cap)
  score = (bins_remain_cap - max_bin_cap)**2 / item + bins_remain_cap**2 / (item**2)
  score += bins_remain_cap**2 / item**3
  score[bins_remain_cap > item] = -score[bins_remain_cap > item]
  score[1:] -= score[:-1]
  return score
